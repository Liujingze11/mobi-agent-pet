# i18n 国际化 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将 DevPulse AI 从中英混合硬编码改造为完整中/英双语国际化系统，首次启动弹窗选语言，设置中可切换。

**Architecture:** React Context (`I18nProvider` + `useI18n` hook) 驱动前端，`electron/i18n.ts` 同步函数驱动主进程。翻译文件为 TS 对象，类型从 `zh-CN.ts` 推导确保中英结构一致。语言设置持久化到 `app_settings.language`。

**Tech Stack:** React 19 Context API, TypeScript template literal types for key paths

---

### 文件结构

```
新建:
  src/lib/i18n/types.ts       — TranslationMap 类型
  src/lib/i18n/zh-CN.ts       — 中文翻译 (~200 keys)
  src/lib/i18n/en-US.ts       — 英文翻译 (同结构)
  src/lib/i18n/index.tsx      — I18nProvider, useI18n, LanguageGate
  electron/i18n.ts            — 主进程 t(db, key) 函数

修改:
  src/gui/App.tsx             — 包裹 I18nProvider + LanguageGate
  src/pulsecore/App.tsx       — 包裹 I18nProvider
  src/components/pulsecore/*  — PulseCore, QuickPanel, ReviewDialog, AchievementToast
  src/components/gui/**/*     — 全部 GUI 组件
  electron/tray.ts            — 动态菜单
  electron/windows.ts         — 动态右键菜单
  electron/ai/deepseek.ts     — 双语 prompt
  electron/db/seed.ts         — 双语种子
  electron/achievements/checker.ts — 双语通知
  src/components/gui/settings/SettingsPage.tsx — 语言选择器
```

```typescript
// src/lib/i18n/index.tsx
import React, { createContext, useContext, useState, useCallback, useEffect } from 'react'
import type { TranslationMap } from './types'
import { zhCN } from './zh-CN'
import { enUS } from './en-US'

export type Language = 'zh-CN' | 'en-US'

const translations: Record<Language, TranslationMap> = {
  'zh-CN': zhCN,
  'en-US': enUS,
}

// 递归获取嵌套属性
function getNestedValue(obj: any, path: string): string {
  const keys = path.split('.')
  let current = obj
  for (const key of keys) {
    if (current === undefined || current === null) return path
    current = current[key]
  }
  return typeof current === 'string' ? current : path
}

interface I18nContextValue {
  language: Language
  setLanguage: (lang: Language) => void
  t: (key: string) => string
  /** 用于翻译数组数据（种子数据等） */
  tData: <T>(key: string) => T
}

const I18nContext = createContext<I18nContextValue>({
  language: 'zh-CN',
  setLanguage: () => {},
  t: (key: string) => key,
  tData: <T,>(key: string) => ({} as T),
})

export function I18nProvider({ children, initialLanguage, onLanguageChange }: {
  children: React.ReactNode
  initialLanguage: Language
  onLanguageChange?: (lang: Language) => void
}) {
  const [language, setLanguageState] = useState<Language>(initialLanguage)

  const setLanguage = useCallback((lang: Language) => {
    setLanguageState(lang)
    onLanguageChange?.(lang)
  }, [onLanguageChange])

  const t = useCallback((key: string): string => {
    const map = translations[language] as any
    return getNestedValue(map, key)
  }, [language])

  const tData = useCallback(<T,>(key: string): T => {
    const map = translations[language] as any
    let current = map
    for (const k of key.split('.')) {
      if (current === undefined || current === null) return undefined as any
      current = current[k]
    }
    return current as T
  }, [language])

  return (
    <I18nContext.Provider value={{ language, setLanguage, t, tData }}>
      {children}
    </I18nContext.Provider>
  )
}

export function useI18n() {
  return useContext(I18nContext)
}

// 首次启动语言选择页 — 替代 mode 选择
export function LanguageGate({ onSelect }: { onSelect: (lang: Language) => void }) {
  return (
    <div className="h-screen flex items-center justify-center bg-surface-dark">
      <div className="text-center space-y-6">
        <h1 className="text-2xl font-bold text-white">
          <span className="text-indigo-400">DevPulse</span> AI
        </h1>
        <p className="text-slate-400">请选择语言 / Choose your language</p>
        <div className="flex gap-4">
          <button
            onClick={() => onSelect('zh-CN')}
            className="bg-indigo-600 hover:bg-indigo-500 text-white rounded-xl px-8 py-4 text-lg font-medium transition-colors"
          >
            🇨🇳 中文
          </button>
          <button
            onClick={() => onSelect('en-US')}
            className="bg-slate-700 hover:bg-slate-600 text-white rounded-xl px-8 py-4 text-lg font-medium transition-colors"
          >
            🇺🇸 English
          </button>
        </div>
      </div>
    </div>
  )
}

// 加载语言设置的 hook
export function useLanguageInit(): { language: Language | null; setAndSave: (lang: Language) => Promise<void> } {
  const [language, setLanguage] = useState<Language | null>(null)

  useEffect(() => {
    const { api } = require('./ipc')
    api.settings.get('language').then((v: string | undefined) => {
      if (v && (v === 'zh-CN' || v === 'en-US')) {
        setLanguage(v as Language)
      }
      // 如果为 null/undefined → language 保持 null → 触发 LanguageGate
    })
    api.app.getMode().then((m: string) => {
      if (!m || m === '""') {
        api.app.setMode('solo')
      }
    })
  }, [])

  const setAndSave = useCallback(async (lang: Language) => {
    const { api } = require('./ipc')
    await api.settings.set('language', lang)
    setLanguage(lang)
  }, [])

  return { language, setAndSave }
}
```

```typescript
export function loadLanguage(db: Database.Database): Language {
```

```typescript
// src/pulsecore/App.tsx — 顶部新增 import，包裹 Provider
import { I18nProvider } from '../lib/i18n'
import type { Language } from '../lib/i18n'

// 在 useState 区域新增:
const [language, setLanguage] = useState<Language>('zh-CN')

// 在 useEffect 中加载语言:
useEffect(() => {
  api.settings.get('language').then(v => {
    if (v === 'zh-CN' || v === 'en-US') setLanguage(v as Language)
  })
}, [])

// return 包裹:
return (
  <I18nProvider initialLanguage={language}>
    <div className="pulsecore-window ..." ...>
      ...
    </div>
  </I18nProvider>
)
```
